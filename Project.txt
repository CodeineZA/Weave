You are a software developer assisting me to build a lightweight webapp for voice chat and screen sharing.
 
The following is a prompt that I have generated:
Build a lightweight, self-hosted voice and screen sharing web app with the following specifications:
## Overview
A minimal Discord-like voice chat room for a small group of trusted friends (8-10 people). No accounts, no sign-up. Just a username, a password, and you're in.
## Core Features

Single persistent room — everyone joins the same space
Voice chat via WebRTC (open mic, no push-to-talk needed)
Screen sharing via WebRTC (getDisplayMedia API)
Username prompt on join (display name only, no account)
Server password protection (one shared password for all users)
See who is currently in the room (list of connected usernames)
Visual indicator when someone is speaking (voice activity detection)
Mute/unmute toggle for own mic
Start/stop screen share toggle
See who is sharing their screen

## Technical Stack

Frontend: Vanilla HTML, CSS, JavaScript (no heavy frameworks)
Signaling server: Node.js with WebSocket (ws library)
WebRTC: Peer-to-peer audio, with STUN/TURN support
STUN: Use Google's free STUN servers (stun.l.google.com:19302)
TURN: Configurable via environment variables (for self-hosting later)
Hosting: Raspberry Pi, exposed via Cloudflare Tunnel

## Architecture

The signaling server handles: room join/leave, WebRTC offer/answer/ICE candidate exchange, broadcasting user list updates
Each client establishes a direct P2P WebRTC connection with every other peer in the room
Audio streams are sent peer-to-peer
Screen share streams are sent peer-to-peer
The signaling server does NOT handle any media — only signaling messages over WebSocket

## Signaling Message Types (WebSocket)
Define and handle these message types:

join — user enters room with username and password
join_ack — server confirms join, sends current user list
join_rejected — wrong password
user_joined — broadcast to others when someone joins
user_left — broadcast when someone disconnects
offer — WebRTC offer from one peer to another
answer — WebRTC answer
ice_candidate — ICE candidate exchange

## UI Requirements

Clean, dark themed minimal interface
On load: show a centered join form (username field, password field, join button)
After joining: show the room view with connected users panel and a controls bar at the bottom
Connected users panel: show each user's avatar (initials-based), username, and a speaking indicator
Controls bar: mute toggle button, screen share toggle button, leave button
If someone is screen sharing, display their stream in the main area
If multiple people share, show all streams in a grid
Responsive — should work on desktop browsers

// Utility: simple hash for avatar color
function getAvatarColor(name) {
    let hash = 0;
    for (let i = 0; i < name.length; i++) {
        hash = name.charCodeAt(i) + ((hash << 5) - hash);
    }
    const colors = ['#00d4ff', '#ff00aa', '#00ff88', '#ffaa00', '#aa88ff'];
    return colors[Math.abs(hash) % colors.length];
}

// Create avatar element
function createAvatar(username) {
    const div = document.createElement('div');
    div.className = 'avatar';
    div.style.backgroundColor = getAvatarColor(username);
    div.textContent = username[0].toUpperCase();
    return div;
}

// Update users list UI
function updateUsersList() {
    usersList.innerHTML = '';
    connectedUsers.forEach(user => {
        if (user === myUsername) return; // we show ourselves only in controls
        
        const item = document.createElement('div');
        item.className = 'user-item';
        item.innerHTML = `
            ${createAvatar(user).outerHTML}
            <span class="username">${user}</span>
            <span id="speaking-${user}" class="speaking" style="display: none;">
                <span class="speaking-dot"></span> speaking
            </span>
        `;
        usersList.appendChild(item);
    });
    
    userCount.textContent = connectedUsers.length;
}

// Add a user to the list
function addUser(username) {
    if (!connectedUsers.includes(username)) {
        connectedUsers.push(username);
        updateUsersList();
    }
}

// Remove a user
function removeUser(username) {
    connectedUsers = connectedUsers.filter(u => u !== username);
    updateUsersList();
    
    // Clean up peer connection
    if (peerConnections.has(username)) {
        const pc = peerConnections.get(username);
        pc.close();
        peerConnections.delete(username);
    }
    
    // Clean up screen
    if (screenVideos.has(username)) {
        screenVideos.get(username).remove();
        screenVideos.delete(username);
    }
    
    // Clean up voice analyser
    if (voiceAnalysers.has(username)) {
        clearInterval(voiceAnalysers.get(username).interval);
        voiceAnalysers.delete(username);
    }
    
    checkEmptyState();
}

// Create RTCPeerConnection for a peer
function createPeerConnection(targetUsername) {
    const pc = new RTCPeerConnection({ iceServers });
    
    peerConnections.set(targetUsername, pc);
    
    // Add local audio track to this peer (if we have it)
    if (localStream) {
        localStream.getTracks().forEach(track => {
            pc.addTrack(track, localStream);
        });
    }
    
    // ICE candidate handling
    pc.onicecandidate = (e) => {
        if (e.candidate && ws && ws.readyState === WebSocket.OPEN) {
            ws.send(JSON.stringify({
                type: 'ice_candidate',
                to: targetUsername,
                candidate: e.candidate
            }));
        }
    };
    
    // Remote track (audio or screen video)
    pc.ontrack = (event) => {
        const track = event.track;
        
        if (track.kind === 'audio') {
            // Play remote audio + start VAD
            const audioEl = document.createElement('audio');
            audioEl.autoplay = true;
            const remoteStream = new MediaStream([track]);
            audioEl.srcObject = remoteStream;
            hiddenAudio.appendChild(audioEl);
            
            startVoiceActivityDetection(targetUsername, remoteStream);
        } 
        else if (track.kind === 'video') {
            // This is a screen share
            const videoStream = new MediaStream([track]);
            addScreenToGrid(targetUsername, videoStream);
            
            // Handle browser stop button on remote side
            track.onended = () => {
                removeScreenFromGrid(targetUsername);
            };
        }
    };
    
    // Connection state monitoring
    pc.onconnectionstatechange = () => {
        if (pc.connectionState === 'failed' || pc.connectionState === 'closed') {
            removeUser(targetUsername);
        }
    };
    
    return pc;
}

// Renegotiate after adding screen track
async function renegotiate(pc, targetUsername) {
    try {
        const offer = await pc.createOffer();
        await pc.setLocalDescription(offer);
        
        if (ws && ws.readyState === WebSocket.OPEN) {
            ws.send(JSON.stringify({
                type: 'offer',
                to: targetUsername,
                offer: pc.localDescription
            }));
        }
    } catch (err) {
        console.error('Renegotiation failed', err);
    }
}

// Add screen to main grid (local or remote)
function addScreenToGrid(username, stream) {
    // Prevent duplicates
    if (screenVideos.has(username)) return;
    
    const container = document.createElement('div');
    container.className = 'screen-container';
    
    const video = document.createElement('video');

    screenVideos.set(username, container);
    
    checkEmptyState();
}

// Remove screen from grid
function removeScreenFromGrid(username) {
    if (screenVideos.has(username)) {
        const container = screenVideos.get(username);
        container.remove();
        screenVideos.delete(username);
        checkEmptyState();
    }
}

// Check if we should show empty state
function checkEmptyState() {
    emptyState.style.display = screenVideos.size === 0 ? 'flex' : 'none';
}

// Voice activity detection (simple volume threshold)
function startVoiceActivityDetection(username, stream) {
    if (voiceAnalysers.has(username)) return;
    
    const audioContext = new (window.AudioContext || window.webkitAudioContext)();
    const source = audioContext.createMediaStreamSource(stream);
    const analyser = audioContext.createAnalyser();
    analyser.fftSize = 512;
    source.connect(analyser);
    
    const speakingEl = document.getElementById(`speaking-${username}`);
    
    const check = () => {
        const dataArray = new Uint8Array(analyser.frequencyBinCount);
        analyser.getByteFrequencyData(dataArray);
        
        let sum = 0;
        for (let i = 0; i < dataArray.length; i++) {
            sum += dataArray[i];
        }
        const average = sum / dataArray.length;
        
        const isSpeakingNow = average > 28; // tuned threshold
        
        if (speakingEl) {
            speakingEl.style.display = isSpeakingNow ? 'flex' : 'none';
        }
    };
    
    const interval = setInterval(check, 180);
    voiceAnalysers.set(username, { context: audioContext, analyser, interval });
}

// Initialize local microphone
async function initLocalMedia() {
    try {
        localStream = await navigator.mediaDevices.getUserMedia({
            audio: {
                echoCancellation: true,
                noiseSuppression: true,
                autoGainControl: true
            }
        });
        return true;
    } catch (err) {
        console.error('Microphone access denied', err);
        alert('Microphone access is required for voice chat.');
        return false;
    }
}

// Connect to signaling server
function connectWebSocket() {
    const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
    ws = new WebSocket(`${protocol}//${window.location.host}`);
    
    ws.onopen = () => {
        reconnectAttempts = 0;
        console.log('✅ WebSocket connected');
        
        // If we already joined (reconnection), re-join automatically
        if (joined && myUsername && roomPassword) {
            console.log('Re-joining after reconnect...');
            ws.send(JSON.stringify({
                type: 'join',
                username: myUsername,
                password: roomPassword
            }));
        }
    };
    
    ws.onmessage = async (event) => {
        const msg = JSON.parse(event.data);
        
        switch (msg.type) {
            case 'join_ack':
                joined = true;
                iceServers = msg.iceServers || [{ urls: 'stun:stun.l.google.com:19302' }];
                connectedUsers = msg.users.map(u => u.username);
                
                // Initialize mic (once)
                if (!localStream) {
                    const ok = await initLocalMedia();
                    if (!ok) return;
                }
                
                myUsernameDisplay.textContent = myUsername;
                
                // Create peer connections to everyone (new joiner initiates)
                for (const user of connectedUsers) {
                    if (user === myUsername) continue;
                    
                    const pc = createPeerConnection(user);
                    
                    // New joiner creates offer
                    try {
                        const offer = await pc.createOffer();
                        await pc.setLocalDescription(offer);
                        ws.send(JSON.stringify({
                            type: 'offer',
                            to: user,
                            offer: pc.localDescription
                        }));
                    } catch (e) {
                        console.error('Offer creation failed', e);
                    }
                }
                
                updateUsersList();
                joinScreen.classList.add('hidden');
                roomScreen.classList.remove('hidden');
                checkEmptyState();
                break;
                
            case 'user_joined':
                addUser(msg.username);
                
                // Existing users create PC for newcomer but do NOT send offer
                if (msg.username !== myUsername) {
                    createPeerConnection(msg.username);
                }
                break;
                
            case 'user_left':
                removeUser(msg.username);
                break;
                
            case 'offer':
                let pc = peerConnections.get(msg.from);
                if (!pc) {
                    pc = createPeerConnection(msg.from);
                }
                
                await pc.setRemoteDescription(msg.offer);
                const answer = await pc.createAnswer();
                await pc.setLocalDescription(answer);
                
                ws.send(JSON.stringify({
                    type: 'answer',
                    to: msg.from,
                    answer: pc.localDescription
                }));
                break;
                
            case 'answer':
                const pcAnswer = peerConnections.get(msg.from);
                if (pcAnswer) {
                    await pcAnswer.setRemoteDescription(msg.answer);
                }
                break;
                
            case 'ice_candidate':
                const pcIce = peerConnections.get(msg.from);
                if (pcIce) {
                    await pcIce.addIceCandidate(msg.candidate);
                }
                break;
                
            case 'join_rejected':
                joinError.textContent = msg.reason || 'Join failed';
                // Allow retry
                break;
        }
    };
    
    ws.onclose = () => {
        console.log('WebSocket closed');
        if (joined && reconnectAttempts < MAX_RECONNECT) {
            reconnectAttempts++;
            console.log(`Reconnecting in ${reconnectAttempts * 2}s...`);
            setTimeout(connectWebSocket, reconnectAttempts * 2000);
        } else if (joined) {
            alert('Connection lost. Please refresh the page.');
        }
    };
    
    ws.onerror = (err) => console.error('WebSocket error', err);
}

// Start screen sharing
async function startScreenShare() {
    try {
        screenStream = await navigator.mediaDevices.getDisplayMedia({
            video: {
                cursor: 'always',
                displaySurface: 'monitor'
            }
        });
        
        const screenTrack = screenStream.getVideoTracks()[0];
        
        // Add screen track to every existing peer connection + renegotiate
        for (const [username, pc] of peerConnections) {
            pc.addTrack(screenTrack, screenStream);
            await renegotiate(pc, username);
        }
        
        // Local preview
        addScreenToGrid(myUsername, screenStream);
        
        isScreenSharing = true;
        screenIcon.textContent = '⏹️';
        screenText.textContent = 'Stop sharing';
        
        // Browser "Stop sharing" button handling
        screenTrack.onended = () => {
            stopScreenShare(true);
        };
        
    } catch (err) {
        console.error('Screen share cancelled', err);
    }
}

// Stop screen sharing
function stopScreenShare(fromBrowser = false) {
    if (screenStream) {
        screenStream.getTracks().forEach(t => t.stop());
        screenStream = null;
    }
    
    removeScreenFromGrid(myUsername);
    
    isScreenSharing = false;
    screenIcon.textContent = '📺';
    screenText.textContent = 'Share screen';
    
    // No renegotiation needed – remote tracks will fire "ended" event naturally
}

// Toggle mute
function toggleMute() {
    if (!localStream) return;
    isMuted = !isMuted;
    localStream.getAudioTracks().forEach(track => {
        track.enabled = !isMuted;
    });
    
    if (isMuted) {
        muteIcon.textContent = '🔇';
        muteText.textContent = 'Unmute';
        muteBtn.classList.add('active');
    } else {
        muteIcon.textContent = '🎤';
        muteText.textContent = 'Mute';
        muteBtn.classList.remove('active');
    }
}

// Leave room
function leaveRoom() {
    if (ws) ws.close();
    joined = false;
    
    // Clean everything
    peerConnections.forEach(pc => pc.close());
    peerConnections.clear();
    screenVideos.forEach(c => c.remove());
    screenVideos.clear();
    voiceAnalysers.forEach(v => clearInterval(v.interval));
    voiceAnalysers.clear();
    
    if (localStream) localStream.getTracks().forEach(t => t.stop());
    if (screenStream) screenStream.getTracks().forEach(t => t.stop());
    
    roomScreen.classList.add('hidden');
    joinScreen.classList.remove('hidden');
    joinError.textContent = '';
    connectedUsers = [];
}

function init() {
    joinBtn.addEventListener('click', async () => {
        const username = usernameInput.value.trim();
        const password = passwordInput.value.trim();
        
        if (!username) {
            joinError.textContent = 'Please enter a name';
            return;
        }
        
        if (!password) {
            joinError.textContent = 'Password required';
            return;
        }
        
        myUsername = username;
        roomPassword = password;
        joinError.textContent = '';
        
        // Connect (or reconnect) WebSocket
        if (!ws || ws.readyState !== WebSocket.OPEN) {
            connectWebSocket();
        } else {
            // Already connected – just send join
            ws.send(JSON.stringify({
                type: 'join',
                username: myUsername,
                password: roomPassword
            }));
        }
    });
    
    // Controls
    muteBtn.addEventListener('click', toggleMute);
    screenBtn.addEventListener('click', () => {
        if (isScreenSharing) {
            stopScreenShare();
        } else {
            startScreenShare();
        }
    });
    
    leaveBtn.addEventListener('click', leaveRoom);
    
    // Keyboard shortcuts
    document.addEventListener('keydown', e => {
        if (e.key === '/' && document.activeElement.tagName !== 'INPUT') {
            e.preventDefault();
            toggleMute();
        }
    });
    
    // Clean up on page unload
    window.addEventListener('beforeunload', () => {
        if (ws) ws.close();
    });
    
    console.log('%c✅ Voice + Screen app ready. Self-hosted and lightweight!', 'color:#00d4ff; font-size:13px');
}

import http from 'http';

import dotenv from 'dotenv';

dotenv.config();

const SERVER_PASSWORD = process.env.SERVER_PASSWORD || 'password123';
const PORT = parseInt(process.env.PORT) || 3000;

const ICE_SERVERS = [
    { urls: 'stun:stun.l.google.com:19302' }
];

if (process.env.TURN_URL) {
    ICE_SERVERS.push({
        urls: process.env.TURN_URL,
        username: process.env.TURN_USERNAME,
        credential: process.env.TURN_CREDENTIAL
    });
}

const users = new Map(); // WebSocket → { username }

const httpServer = http.createServer((req, res) => {
    let filePath = path.join(process.cwd(), 'public', req.url === '/' ? 'index.html' : req.url);
    
    const ext = path.extname(filePath).toLowerCase();
    let contentType = 'text/html';
    if (ext === '.css') contentType = 'text/css';
    else if (ext === '.js') contentType = 'text/javascript';

    fs.readFile(filePath, (err, data) => {
        if (err) {
            res.writeHead(404);
            res.end('Not found');
            return;
        }
        res.writeHead(200, { 'Content-Type': contentType });
        res.end(data);
    });
});

const wss = new WebSocketServer({ server: httpServer });

wss.on('connection', (ws) => {
    console.log('🔌 Client connected');

    ws.on('message', (raw) => {
        let msg;
        try {
            msg = JSON.parse(raw);
        } catch (e) {
            return;
        }

        if (msg.type === 'join') {
            const { username, password } = msg;

            if (password !== SERVER_PASSWORD) {
                ws.send(JSON.stringify({ type: 'join_rejected', reason: 'Wrong password' }));
                return;
            }

            // Username must be unique
            const taken = Array.from(users.values()).some(u => u.username === username);
            if (taken) {
                ws.send(JSON.stringify({ type: 'join_rejected', reason: 'Username already taken' }));
                return;
            }

            users.set(ws, { username });

            // Send full user list + ICE servers to new client
            const userList = Array.from(users.values()).map(u => ({ username: u.username }));
            ws.send(JSON.stringify({
                type: 'join_ack',
                users: userList,
                iceServers: ICE_SERVERS
            }));

            // Broadcast to everyone else
            const joinedMsg = JSON.stringify({ type: 'user_joined', username });
            for (const [client] of users) {
                if (client !== ws && client.readyState === 1) {
                    client.send(joinedMsg);
                }
            }
        }
        else if (['offer', 'answer', 'ice_candidate'].includes(msg.type)) {
            if (!users.has(ws)) return; // not joined yet

            const from = users.get(ws).username;
            const targetWs = Array.from(users.entries())
                .find(([_, u]) => u.username === msg.to)?.[0];

            if (targetWs && targetWs.readyState === 1) {
                const forward = { ...msg, from };
                targetWs.send(JSON.stringify(forward));
            }
        }
    });

    ws.on('close', () => {
        if (users.has(ws)) {
            const { username } = users.get(ws);
            users.delete(ws);

            const leftMsg = JSON.stringify({ type: 'user_left', username });
            for (const [client] of users) {
                if (client.readyState === 1) client.send(leftMsg);
            }
            console.log(`👋 ${username} left`);
        }
    });
});

httpServer.listen(PORT, () => {
